   🤖 بوت أب — منصة إنشاء بوتات "الناس تحكيني"
═══════════════════════════════════════════════

شو بيعمل:
  عندك بوت رئيسي واحد. أي شخص بيفوت عليه، بيحط توكن
  بوته (من BotFather)، وبصير عندو تلقائياً بوت خاص فيه
  "الناس تحكيني" بكل الميزات:
   • الرسائل بتوصله مع اسم وصورة المرسِل
   • بيرد عليهم بالرد (reply)
   • حظر / كتم / منع روابط
  كلو مجاني، وكل بوت مستقل.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
خطوة 1: انشئ البوت الرئيسي
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• روح عند @BotFather → /newbot
• خد التوكن تبع البوت الرئيسي (هدا اللي رح تستضيفه)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
خطوة 2: ارفع الملفات على GitHub
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ارفع (أو بدّل) هالملفات بمستودعك:
   • bot.py
   • requirements.txt
   • runtime.txt

⚠️ امسح الملفات القديمة (bot.py القديم و Procfile
   ما عاد لهم لزوم) وحط هدول مكانهم.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
خطوة 3: على Render (نفس اللي عملته)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• نفس الـ Web Service المجاني
• Build Command:  pip install -r requirements.txt
• Start Command:  python bot.py
• بتبويب Environment، ضيف متغير:
     الاسم:  OWNER_BOT_TOKEN
     القيمة: توكن البوت الرئيسي
   (هيك التوكن ما بيكون مكشوف بالكود — أأمن)
• احفظ، وRender رح يعيد النشر لحاله.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
كيف يستعملها الناس
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1) الشخص بيفوت عالبوت الرئيسي ويكتب /start
2) البوت بيقلّه: اعمل بوت من BotFather وابعتلي توكنه
3) الشخص بيبعت التوكن → البوت بيشغّله فوراً
4) بصير عندو بوت خاص، الناس بتبعتله وبتوصله الرسائل


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ملاحظات مهمة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• البيانات بتنحفظ بملف botup_data.json. بس انتبه:
  على خطة Render المجانية، الملف بيتمسح مع كل
  إعادة نشر (redeploy). البوتات بترجع تشتغل لأن
  التوكنات محفوظة، بس لو بدك حفظ دائم 100% لازم
  قاعدة بيانات (منعملها لاحقاً لو حبيت).
• الخطة المجانية بتنام لو ما في نشاط، وبتصحى لما
  تجي رسالة (ثواني). عادي للاستعمال الطبيعي.
• كل بوت جديد بياخد شوية موارد. المجاني بيكفي
  لعدد محدود من البوتات. لو صاروا كتير، بتحتاج
  ترقّي الخطة.

تشغيل محلي (اختياري، للتجربة)

   export OWNER_BOT_TOKEN="توكن_البوت_الرئيسي"
